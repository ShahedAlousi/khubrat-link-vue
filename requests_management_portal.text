            if (tab === 'pending') {
                pendingBtn.className = "px-4 py-2 text-xs font-bold rounded-lg bg-khubrat-blue text-white dark:bg-khubrat-goldLight dark:text-khubrat-blue transition-all shadow-sm";
                historyBtn.className = "px-4 py-2 text-xs font-semibold text-slate-500 dark:text-slate-400 rounded-lg transition-all";
            } else {
                historyBtn.className = "px-4 py-2 text-xs font-bold rounded-lg bg-khubrat-blue text-white dark:bg-khubrat-goldLight dark:text-khubrat-blue transition-all shadow-sm";
                pendingBtn.className = "px-4 py-2 text-xs font-semibold text-slate-500 dark:text-slate-400 rounded-lg transition-all";
            }

            filterAndRenderRequests();
        }

        // VIEW STYLE SWITCHER (GRID VS LIST)
        function setRequestsViewStyle(style) {
            requestsViewStyle = style;
            const container = document.getElementById('requests-listing-container');
            const listBtn = document.getElementById('req-view-btn-list');
            const gridBtn = document.getElementById('req-view-btn-grid');

            if (style === 'grid') {
                container.className = "grid grid-cols-1 md:grid-cols-2 lg:grid-cols-2 gap-4 transition-all duration-300";
                gridBtn.className = "p-2 text-xs rounded-lg transition-all bg-white dark:bg-slate-800 shadow-sm text-khubrat-blue dark:text-khubrat-goldLight";
                listBtn.className = "p-2 text-xs rounded-lg transition-all text-slate-400";

                container.className = "grid grid-cols-1 gap-4 transition-all duration-300";
                listBtn.className = "p-2 text-xs rounded-lg transition-all bg-white dark:bg-slate-800 shadow-sm text-khubrat-blue dark:text-khubrat-goldLight";
                gridBtn.className = "p-2 text-xs rounded-lg transition-all text-slate-400";
            }

            filterAndRenderRequests();
        }

        // FILTER AND RENDER REQUESTS
        function filterAndRenderRequests() {
            const container = document.getElementById('requests-listing-container');
            const selectedType = document.getElementById('req-type-filter').value;

            const filtered = requests.filter(req => {
                const matchesTab = activeRequestsTab === 'pending' ? req.status === 'pending' : req.status !== 'pending';
                const matchesType = selectedType === 'all' || req.type === selectedType;
                return matchesTab && matchesType;
            });

            if (filtered.length === 0) {
                container.innerHTML = `
                    <div class="bg-white dark:bg-slate-800 p-12 rounded-2xl border border-slate-200 dark:border-slate-700 text-center space-y-3">
                        <div class="w-16 h-16 rounded-full bg-slate-100 dark:bg-slate-700/50 flex items-center justify-center mx-auto text-slate-400 text-2xl">
                            <i class="fa-solid fa-folder-open"></i>
                        </div>
                        <h4 class="text-sm font-bold text-slate-700 dark:text-slate-200">No requests match this view</h4>
                        <p class="text-xs text-slate-400 max-w-sm mx-auto">There are currently no ${activeRequestsTab === 'pending' ? 'pending' : 'resolved'} requests for the chosen filter criteria.</p>
                    </div>
                `;
                return;
            }

            container.innerHTML = filtered.map(req => renderRequestItem(req)).join('');
        }

        function renderRequestItem(req) {
            const typeBadges = {
                'advance': '<span class="px-2.5 py-1 rounded-lg text-[10px] font-black bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border border-emerald-500/20"><i class="fa-solid fa-hand-holding-dollar mr-1"></i> Salary Advance</span>',
                'leave': '<span class="px-2.5 py-1 rounded-lg text-[10px] font-black bg-blue-500/10 text-blue-600 dark:text-blue-400 border border-blue-500/20"><i class="fa-solid fa-plane-departure mr-1"></i> Leave Request</span>',

                            <div class="flex items-center gap-3">
                                <img src="${req.employeeAvatar}" alt="${req.employeeName}" class="w-10 h-10 rounded-xl bg-slate-100 dark:bg-slate-700">

                                    <h4 class="text-sm font-bold text-slate-800 dark:text-slate-100">${req.employeeName}</h4>

                        </div>

        function closeRequestDrawer() {
            document.getElementById('request-drawer').classList.add('hidden');
        }

        // DECISION ACTIONS LOGIC
        function approveRequest(id) {
            const req = requests.find(r => r.id === id);
            if (!req) return;

            req.status = 'approved';
            closeRequestDrawer();
            filterAndRenderRequests();
            recalculateRequestsStats();
            showToast(`Request for ${req.employeeName} has been approved successfully!`, 'success');
        }

        function openRejectionModal(id) {
            document.getElementById('rejection-target-id').value = id;
            document.getElementById('rejection-text-input').value = '';
            document.getElementById('rejection-modal').classList.remove('hidden');
        }

        function closeRejectionModal() {
            document.getElementById('rejection-modal').classList.add('hidden');
        }

        function applyRejectionSnippet(text) {
            document.getElementById('rejection-text-input').value = text;
        }

        function submitRejectionAction() {
            const id = document.getElementById('rejection-target-id').value;
            const reason = document.getElementById('rejection-text-input').value.trim();

            if (!reason) {
                showToast('Please provide or choose a reason for rejection.', 'error');
                return;
            }

            const req = requests.find(r => r.id === id);
            if (!req) return;

            req.status = 'rejected';
            req.rejectionReason = reason;

            closeRejectionModal();
            closeRequestDrawer();
            filterAndRenderRequests();
            recalculateRequestsStats();
            showToast(`Request for ${req.employeeName} has been rejected.`, 'error');
        }

        function recalculateRequestsStats() {
            const pendingCount = requests.filter(r => r.status === 'pending').length;
            const resolvedCount = requests.filter(r => r.status !== 'pending').length;

            document.getElementById('stats-pending-count').innerText = pendingCount;
            document.getElementById('stats-resolved-count').innerText = resolvedCount;
        }

        // TOAST NOTIFICATIONS HELPER
        function showToast(message, type = 'info') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');

            let bgStyle = 'bg-slate-800 text-white';
            if (type === 'success') bgStyle = 'bg-emerald-600 text-white';
            if (type === 'error') bgStyle = 'bg-rose-600 text-white';

            toast.className = `px-4 py-3 rounded-xl shadow-xl font-bold text-xs flex items-center gap-2 transition-all transform translate-y-2 animate-in fade-in ${bgStyle}`;
            toast.innerHTML = `<span>${message}</span>`;

            container.appendChild(toast);

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }
